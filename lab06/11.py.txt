A = list(map(int, input("Nhap danh sach: ").split()))

# a
B = [x for x in A if x % 3 == 0 and x % 5 != 0]

print("B =", B)

# b
C = [x**2 for x in A]

print("C =", C)

# c
import random

D = [x for x in A if x % 3 == 0]

if len(D) > 0:
    print("Ngau nhien:", random.choice(D))
else:
    print("Khong co phan tu chia het cho 3")
