# Ham kiem tra nam nhuan
def nam_nhuan(y):
    return (y % 400 == 0) or (y % 4 == 0 and y % 100 != 0)

# Ham tinh so ngay trong thang
def so_ngay(thang, nam):
    if thang in [1, 3, 5, 7, 8, 10, 12]:
        return 31
    elif thang in [4, 6, 9, 11]:
        return 30
    elif thang == 2:
        if nam_nhuan(nam):
            return 29
        else:
            return 28
    else:
        return -1

thang = int(input("Nhap thang: "))
nam = int(input("Nhap nam: "))

if nam_nhuan(nam):
    print(nam, "la nam nhuan")
else:
    print(nam, "khong phai nam nhuan")

print("So ngay cua thang", thang, "la:", so_ngay(thang, nam))
